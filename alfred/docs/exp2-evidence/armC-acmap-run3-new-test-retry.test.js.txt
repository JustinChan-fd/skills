// retry — shared policy: 3 attempts, exponential backoff.

import { test } from 'node:test';
import assert from 'node:assert/strict';

import { MAX_ATTEMPTS, backoffMs } from '../src/retry.js';

test('the shared policy allows exactly three attempts', () => {
  assert.equal(MAX_ATTEMPTS, 3);
});

test('backoffMs grows exponentially between attempts', () => {
  const first = backoffMs(1);
  const second = backoffMs(2);
  const third = backoffMs(3);

  assert.ok(second > first);
  assert.ok(third > second);
  assert.equal(second / first, third / second);
});
